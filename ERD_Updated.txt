    User ||--o{ UserRole : has
    User ||--o{ Session : has
    User ||--o{ Log : acts
    User ||--o| InfluencerProfile : has
    User ||--o| OwnerProfile : has
    User ||--o{ Notification : receives
    User ||--o{ UploadedFile : uploads
    
    Role ||--o{ UserRole : has
    
    %% ========================================
    %% CAMPAIGNS & AI
    %% ========================================
    
    User ||--o{ Campaign : owns
    Campaign ||--o{ KPI : has
    Campaign ||--|| TargetAudience : has
    Campaign ||--o{ ContentCalendar : has
    Campaign ||--o{ CampaignAIVersion : has
    
    %% ========================================
    %% COLLABORATIONS WORKFLOW
    %% ========================================
    
    Campaign ||--o{ CollaborationRequest : has
    User ||--o{ CollaborationRequest : "sends (as owner)"
    User ||--o{ CollaborationRequest : "receives (as influencer)"
    
    CollaborationRequest ||--o| Collaboration : "creates when accepted"
    
    Campaign ||--o{ Collaboration : has
    User ||--o{ Collaboration : "participates (as owner)"
    User ||--o{ Collaboration : "participates (as influencer)"
    
    Collaboration ||--o| CollaborationContract : has
    Collaboration ||--o{ CollaborationTask : has
    Collaboration ||--o| ChatRoom : "may have"
    
    %% ========================================
    %% CHAT SYSTEM
    %% ========================================
    
    ChatRoom ||--o{ ChatParticipant : has
    ChatRoom ||--o{ Message : contains
    User ||--o{ ChatParticipant : participates
    User ||--o{ Message : sends
    
    %% ========================================
    %% SERVICE MARKETPLACE
    %% ========================================
    
    User ||--o{ ServiceListing : "posts (as influencer)"
    User ||--o{ ServiceRequest : "posts (as owner)"
    
    ServiceListing ||--o{ Offer : receives
    ServiceRequest ||--o{ Proposal : receives
    
    User ||--o{ Offer : "makes (as owner)"
    User ||--o{ Proposal : "makes (as influencer)"

    %% ========================================
    %% TABLE DEFINITIONS
    %% ========================================

    User {
        int id PK
        string firstName
        string lastName
        string email UK
        string password "nullable OAuth users"
        string googleId UK "nullable"
        enum status "ACTIVE BLOCKED SUSPENDED INCOMPLETE"
        string resetPasswordToken "nullable"
        date resetPasswordExpires "nullable"
        timestamp createdAt
        timestamp updatedAt
    }

    Role {
        int id PK
        enum name UK "OWNER INFLUENCER ADMIN"
        timestamp createdAt
        timestamp updatedAt
    }

    UserRole {
        int id PK
        int userId FK
        int roleId FK
        timestamp createdAt
        timestamp updatedAt
    }

    Session {
        int id PK
        int userId FK
        string refreshTokenHash
        string ip "nullable"
        text userAgent "nullable"
        date expiresAt
        date revokedAt "nullable"
        timestamp createdAt
    }

    Log {
        int id PK
        int actorId FK "nullable"
        string actor
        string action
        string entity
        int entityId "nullable"
        json meta "nullable"
        timestamp createdAt
    }

    InfluencerProfile {
        int id PK
        int userId FK "unique"
        text bio "nullable"
        string image "nullable"
        string location "nullable"
        boolean isCompleted "default false"
        jsonb socialMediaLinks "default {}"
        string primaryPlatform "nullable"
        int followersCount "default 0"
        decimal engagementRate "default 0.0"
        array categories "nullable"
        array contentTypes "nullable"
        array collaborationTypes "nullable"
        string audienceAgeRange "nullable"
        string audienceGender "nullable"
        string audienceLocation "nullable"
        array interests "nullable"
        int completionPercentage "default 0"
        boolean isOnboarded "default false"
        timestamp createdAt
        timestamp updatedAt
    }

    OwnerProfile {
        int id PK
        int userId FK "unique"
        string businessName "nullable"
        string businessType "nullable"
        string industry "nullable"
        string location "nullable"
        text description "nullable"
        string image "nullable"
        string website "nullable"
        string phoneNumber "nullable"
        array platformsUsed "nullable"
        string primaryMarketingGoal "nullable"
        jsonb targetAudience "nullable"
        int completionPercentage "default 0"
        boolean isOnboarded "default false"
        boolean isCompleted "default false"
        timestamp createdAt
        timestamp updatedAt
    }

    Brand {
        int id PK
        int ownerId FK
        string name UK
        string slug UK
        text mission "nullable"
        text vision "nullable"
        json values "default []"
        json targetAudience "default {}"
        json productsAndServices "default []"
        timestamp createdAt
        timestamp updatedAt
    }

    Campaign {
        int id PK
        int userId FK
        string campaignName
        enum lifecycleStage "draft ai_generated saved completed cancelled"
        text UserDescription
        enum goalType "awareness consideration conversion lead_generation retention"
        float totalBudget
        string currency
        enum budgetFlexibility "strict flexible"
        boolean isPublished "default false"
        date startDate
        date endDate
        timestamp createdAt
        timestamp updatedAt
    }

    KPI {
        int id PK
        int campaignId FK
        enum metric "impressions reach engagement_rate conversions ROAS CPA CTR"
        string targetValue
        timestamp createdAt
        timestamp updatedAt
    }

    TargetAudience {
        int id PK
        int campaignId FK
        string ageRange "nullable"
        enum gender "all male female custom"
        array interests "nullable"
        array platformsUsed "nullable"
        timestamp createdAt
        timestamp updatedAt
    }

    ContentCalendar {
        int id PK
        int campaignId FK
        int day
        date date
        string platform
        enum contentType "video carousel story reel post article"
        text caption "nullable"
        string mediaUrl "nullable"
        string task "nullable"
        enum status "scheduled posted failed"
        timestamp createdAt
        timestamp updatedAt
    }

    CampaignAIVersion {
        uuid id PK
        int campaignId FK
        int versionNumber
        timestamp generatedAt
        json strategy "nullable"
        json execution "nullable"
        json estimations "nullable"
        boolean isActive "default false"
        timestamp createdAt
        timestamp updatedAt
    }

    CollaborationRequest {
        int id PK
        int campaignId FK
        int ownerId FK
        int influencerId FK
        enum status "pending negotiating accepted rejected cancelled expired"
        decimal proposedBudget "nullable"
        decimal counterPrice "nullable"
        text message "nullable"
        text responseMessage "nullable"
        date expiresAt "nullable"
        timestamp createdAt
        timestamp updatedAt
    }

    Collaboration {
        int id PK
        int collaborationRequestId FK
        int campaignId FK
        int ownerId FK
        int influencerId FK
        enum status "pending_contract_sign live in_progress completed cancelled"
        date startDate "nullable"
        date endDate "nullable"
        date completedAt "nullable"
        date cancelledAt "nullable"
        timestamp createdAt
        timestamp updatedAt
    }

    CollaborationContract {
        int id PK
        int collaborationId FK "unique"
        decimal agreedPrice
        json deliverables "default []"
        date startDate "nullable"
        date endDate "nullable"
        enum status "draft sent signed"
        text notes "nullable"
        timestamp createdAt
        timestamp updatedAt
    }

    CollaborationTask {
        int id PK
        int collaborationId FK
        string taskName
        text description "nullable"
        enum status "todo in_progress in_review approved rejected"
        int sortOrder "default 0"
        date dueDate "nullable"
        date completedAt "nullable"
        enum platform "instagram tiktok youtube facebook twitter linkedin snapchat whatsapp other"
        enum contentType "post story reel video carousel article tweet poll"
        text submissionNote "nullable"
        date submittedAt "nullable"
        text reviewNote "nullable"
        timestamp createdAt
        timestamp updatedAt
    }

    ChatRoom {
        int id PK
        int collaborationId FK "nullable"
        enum type "one_to_one group"
        string name "nullable for group chats"
        timestamp createdAt
    }

    ChatParticipant {
        int id PK
        int chatRoomId FK
        int userId FK
        enum role "owner influencer admin"
        date joinedAt
        timestamp createdAt
        timestamp updatedAt
    }

    Message {
        int id PK
        int chatRoomId FK
        int senderId FK
        text content
        date sentAt
        enum status "sent delivered read"
        timestamp createdAt
        timestamp updatedAt
    }

    ServiceListing {
        int id PK
        int influencerId FK
        string title
        text description
        array categories "default []"
        array platforms "default []"
        string location "nullable"
        decimal price
        enum status "draft published archived"
        timestamp createdAt
        timestamp updatedAt
    }

    ServiceRequest {
        int id PK
        int ownerId FK
        string title
        text description
        array requiredCategories "nullable"
        array requiredPlatforms "nullable"
        string location "nullable"
        decimal budget
        enum status "draft published closed cancelled"
        timestamp createdAt
        timestamp updatedAt
    }

    Offer {
        int id PK
        int serviceListingId FK
        int ownerId FK
        decimal offerPrice
        text message "nullable"
        enum status "pending negotiated accepted rejected expired withdrawn"
        timestamp createdAt
        timestamp updatedAt
    }

    Proposal {
        int id PK
        int serviceRequestId FK
        int influencerId FK
        decimal proposedPrice
        text message "nullable"
        enum status "pending accepted rejected"
        timestamp createdAt
        timestamp updatedAt
    }

    Notification {
        int id PK
        int userId FK
        string message
        enum type "info warning error"
        boolean isRead "default false"
        timestamp createdAt
        timestamp updatedAt
    }

    UploadedFile {
        int id PK
        int userId FK
        string fileName
        string originalName
        string mimeType
        int size
        string path
        string url "nullable"
        enum type "image document"
        boolean isPublic "default false"
        timestamp createdAt
        timestamp updatedAt
    }
