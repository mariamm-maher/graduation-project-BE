    User ||--o{ Notification : receives

    %% ========================================
    %% TABLE DEFINITIONS (SYNCED WITH MODELS)
    %% ========================================

    User {
        int id PK
        string firstName
        string lastName
        string email UK
        string password "nullable OAuth users"
        string googleId UK "nullable"
        enum status "ACTIVE BLOCKED SUSPENDED INCOMPLETE"
        string resetPasswordToken "nullable"
        date resetPasswordExpires "nullable"
        timestamp createdAt
        timestamp updatedAt
    }

    Role {
        int id PK
        enum name UK "OWNER INFLUENCER ADMIN"
        timestamp createdAt
        timestamp updatedAt
    }

    UserRole {
        int id PK
        int userId FK
        int roleId FK
        timestamp createdAt
        timestamp updatedAt
    }

    Session {
        int id PK
        int userId FK
        string refreshTokenHash
        string ip "nullable"
        text userAgent "nullable"
        date expiresAt
        date revokedAt "nullable"
        timestamp createdAt
    }

    Log {
        int id PK
        int actorId FK "nullable"
        string actor "nullable"
        string action
        string entity
        int entityId "nullable"
        json meta "nullable"
        timestamp createdAt
    }

    InfluencerProfile {
        int id PK
        int userId FK "unique"
        text bio "nullable"
        string image "nullable"
        string location "nullable"
        boolean isCompleted "default false"
        jsonb socialMediaLinks "default {}"
        string primaryPlatform "nullable"
        string followersCount "nullable"
        string engagementRate "nullable"
        array categories "nullable"
        array contentTypes "nullable"
        array collaborationTypes "nullable"
        string audienceAgeRange "nullable"
        string audienceGender "nullable"
        string audienceLocation "nullable"
        array interests "nullable"
        int completionPercentage "default 0"
        boolean isOnboarded "default false"
        timestamp createdAt
        timestamp updatedAt
    }

    OwnerProfile {
        int id PK
        int userId FK "unique"
        string businessName "nullable"
        string businessType "nullable"
        string industry "nullable"
        string location "nullable"
        text description "nullable"
        string image "nullable"
        string website "nullable"
        string phoneNumber "nullable"
        array platformsUsed "nullable"
        string primaryMarketingGoal "nullable"
        jsonb targetAudience "nullable"
        int completionPercentage "default 0"
        boolean isOnboarded "default false"
        boolean isCompleted "default false"
        timestamp createdAt
        timestamp updatedAt
    }

    Brand {
        int id PK
        int ownerId FK
        string name UK
        string slug UK
        text mission "nullable"
        text vision "nullable"
        json values "nullable"
        json targetAudience "nullable"
        json productsAndServices "nullable"
        boolean isActive "default true"
        timestamp createdAt
        timestamp updatedAt
    }

    Campaign {
        int id PK
        int userId FK
        string campaignName
        enum lifecycleStage "draft ai_generated saved completed cancelled"
        text UserDescription
        enum goalType "awareness consideration conversion lead_generation retention"
        float totalBudget
        string currency
        enum budgetFlexibility "strict flexible"
        boolean isPublished "default false"
        date startDate
        date endDate
        timestamp createdAt
        timestamp updatedAt
    }

    KPI {
        int id PK
        int campaignId FK
        enum metric "impressions reach engagement_rate conversions ROAS CPA CTR"
        string targetValue
    }

    TargetAudience {
        int id PK
        int campaignId FK
        string ageRange
        enum gender "all male female custom"
        array interests
        array platformsUsed
    }

    ContentCalendar {
        int id PK
        int campaignId FK
        int day
        date date
        string platform
        enum contentType "video carousel story reel post article"
        text caption
        string mediaUrl "nullable"
        string task
        enum status "scheduled posted failed"
    }

    CampaignAIVersion {
        int id PK
        int campaignId FK
        int versionNumber
        timestamp generatedAt "nullable"
        jsonb strategy "nullable"
        jsonb execution "nullable"
        jsonb estimations "nullable"
        boolean isActive "default false"
    }

    CollaborationRequest {
        int id PK
        int campaignId FK
        int ownerId FK
        int influencerId FK
        enum status "pending negotiating accepted rejected cancelled expired"
        decimal proposedBudget "nullable"
        decimal counterPrice "nullable"
        int lastCounteredBy "nullable"
        text message "nullable"
        text responseMessage "nullable"
        date expiresAt "nullable"
        timestamp createdAt
        timestamp updatedAt
    }

    Collaboration {
        int id PK
        int collaborationRequestId FK
        int campaignId FK
        int ownerId FK
        int influencerId FK
        enum status "pending_contract_sign live in_progress completed cancelled"
        date startDate "nullable"
        date endDate "nullable"
        timestamp completedAt "nullable"
        timestamp cancelledAt "nullable"
        timestamp createdAt
        timestamp updatedAt
    }

    CollaborationContract {
        int id PK
        int collaborationId FK "unique"
        decimal agreedPrice
        json deliverables "default []"
        date startDate "nullable"
        date endDate "nullable"
        enum status "sent partially_signed signed cancelled"
        boolean ownerSigned "default false"
        boolean influencerSigned "default false"
        timestamp ownerSignedAt "nullable"
        timestamp influencerSignedAt "nullable"
        string contractFileUrl "nullable"
        text notes "nullable"
        timestamp createdAt
        timestamp updatedAt
    }

    CollaborationTask {
        int id PK
        int collaborationId FK
        string taskName
        text description "nullable"
        enum status "todo in_progress in_review approved rejected"
        int sortOrder "default 0"
        date dueDate "nullable"
        timestamp completedAt "nullable"
        enum platform "instagram tiktok youtube facebook twitter linkedin snapchat whatsapp other"
        enum contentType "post story reel video carousel article tweet poll"
        text submissionNote "nullable"
        timestamp submittedAt "nullable"
        text reviewNote "nullable"
        timestamp createdAt
        timestamp updatedAt
    }

    ChatRoom {
        int id PK
        int collaborationId FK "nullable"
        enum type "one_to_one group"
        string name "nullable for group chats"
        timestamp createdAt
    }

    ChatParticipant {
        int id PK
        int chatRoomId FK
        int userId FK
        enum role "owner influencer admin"
        timestamp joinedAt
    }

    Message {
        int id PK
        int chatRoomId FK
        int senderId FK
        text content
        timestamp sentAt
        enum status "sent delivered read"
    }

    Notification {
        int id PK
        int userId FK
        enum type "..more"
        text message
        string entityType "nullable"
        int entityId "nullable"
        jsonb metadata "nullable"
        boolean isRead "default false"
        timestamp readAt "nullable"
        timestamp createdAt
        timestamp updatedAt
    }

    SocialMediaAccount {
        int id PK
        int userId FK
        enum platform "instagram facebook twitter linkedin tiktok youtube"
        string accountId
        string accountName "nullable"
        string accountUsername "nullable"
        string profilePicture "nullable"
        text accessToken
        text refreshToken "nullable"
        timestamp tokenExpiresAt "nullable"
        jsonb platformData "default {}"
        boolean isActive "default true"
        timestamp lastSyncAt "nullable"
        timestamp createdAt
        timestamp updatedAt
    }

    ScheduledPost {
        int id PK
        int userId FK
        int socialMediaAccountId FK
        int campaignId FK "nullable"
        int contentCalendarId FK "nullable"
        enum platform "instagram facebook twitter linkedin tiktok youtube"
        enum contentType "post story reel video carousel article tweet poll"
        text content
        array mediaUrls "default []"
        timestamp scheduledAt
        enum status "draft scheduled publishing published failed"
        timestamp publishedAt "nullable"
        string platformPostId "nullable"
        text errorMessage "nullable"
        int retryCount "default 0"
        jsonb options "default {}"
        timestamp createdAt
        timestamp updatedAt
    }
